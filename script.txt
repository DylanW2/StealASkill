local function n(value, fallback, minValue, maxValue)
	local number = tonumber(value)
	if number == nil then
		number = fallback
	end

	if minValue ~= nil then
		number = math.max(minValue, number)
	end

	if maxValue ~= nil then
		number = math.min(maxValue, number)
	end

	return number
end

local function waitUntil(timeout, predicate)
	local start = os.clock()

	while os.clock() - start < timeout do
		local ok, result = pcall(predicate)
		if ok and result then
			return true
		end

		task.wait(0.04)
	end

	return false
end

local function isTargetAllowed(player)
	if not REQUIRE_TARGET_CONSENT then
		return true
	end

	if not player then
		return false
	end

	if player == LocalPlayer then
		return true
	end

	if player:GetAttribute(CONSENT_ATTRIBUTE) == true then
		return true
	end

	local character = player.Character
	if character then
		if character:GetAttribute(CONSENT_ATTRIBUTE) == true then
			return true
		end

		if character:GetAttribute(TARGET_ATTRIBUTE) == true then
			return true
		end
	end

	return false
end

local function getTargetParts()
	if not selectedPlayer then
		return nil, "No selected player"
	end

	local character = selectedPlayer.Character
	if not character then
		return nil, "Target has no character"
	end

	local humanoid = character:FindFirstChild("Humanoid")
	local hrp = character:FindFirstChild("HumanoidRootPart")
	local stats = character:FindFirstChild("CharcterStats")

	if not humanoid then
		return nil, "Target missing Humanoid"
	end

	if not hrp then
		return nil, "Target missing HumanoidRootPart"
	end

	if not stats then
		return nil, "Target missing CharcterStats"
	end

	local stun = stats:FindFirstChild("Stun")
	local ragdoll = stats:FindFirstChild("IsRagdoll")
	local cantHit = stats:FindFirstChild("CantHit")

	if not stun then
		return nil, "Target missing Stun"
	end

	if not ragdoll then
		return nil, "Target missing IsRagdoll"
	end

	if not cantHit then
		return nil, "Target missing CantHit"
	end

	return {
		Character = character,
		Humanoid = humanoid,
		HRP = hrp,
		Stats = stats,
		Stun = stun,
		Ragdoll = ragdoll,
		CantHit = cantHit,
	}, nil
end

local function fireServerFunction(functionName, args)
	if not MainEvent then
		return false, "MainEvent missing"
	end

	MainEvent:FireServer(REMOTE_CODE, functionName, args)
	return true
end

local function getTimeout()
	return n(DEFAULTS.Timeout, DEFAULTS.Timeout, CAPS.MinTimeout, CAPS.MaxTimeout)
end

--// Draggable top bar

do
	local dragging = false
	local dragStart
	local startPos

	top.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not dragging then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local delta = input.Position - dragStart

		frame.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end)
end

local resultData = {}

local function addResult(name, state, detail)
	local color = Color3.fromRGB(70, 80, 110)

	if state == "VULNERABLE" then
		color = Color3.fromRGB(175, 55, 55)
	elseif state == "BLOCKED" then
		color = Color3.fromRGB(45, 120, 70)
	elseif state == "SKIPPED" then
		color = Color3.fromRGB(120, 105, 45)
	end

	local row = Instance.new("TextLabel")
	row.Size = UDim2.new(1, -4, 0, 38)
	row.BackgroundColor3 = Color3.fromRGB(38, 38, 48)
	row.TextColor3 = Color3.fromRGB(235, 235, 245)
	row.TextSize = 11
	row.Font = Enum.Font.Gotham
	row.TextXAlignment = Enum.TextXAlignment.Left
	row.TextYAlignment = Enum.TextYAlignment.Center
	row.Text = string.format("  [%s] %s\n  %s", state, name, detail or "")
	row.Parent = results

	local rowCorner = Instance.new("UICorner")
	rowCorner.CornerRadius = UDim.new(0, 7)
	rowCorner.Parent = row

	local stripe = Instance.new("Frame")
	stripe.Size = UDim2.new(0, 4, 1, 0)
	stripe.BackgroundColor3 = color
	stripe.BorderSizePixel = 0
	stripe.Parent = row

	table.insert(resultData, {
		Name = name,
		State = state,
		Detail = detail,
	})

	results.CanvasPosition = Vector2.new(0, math.max(0, results.CanvasSize.Y.Offset))
end

local function blocked(name, detail)

end

local function vuln(name, detail)

end

	addResult(name, "SKIPPED", detail)
end

local function clearResults()
	for _, child in ipairs(results:GetChildren()) do
		if child:IsA("TextLabel") then
			child:Destroy()
		end
	end

	table.clear(resultData)
	setStatus("Cleared", Color3.fromRGB(35, 35, 45))
end

local function buildReport()
	local v, b, s = 0, 0, 0

	for _, item in ipairs(resultData) do
		if item.State == "VULNERABLE" then
			v += 1
		elseif item.State == "BLOCKED" then
			b += 1
		elseif item.State == "SKIPPED" then
			s += 1
		end
	end

	local lines = {}

	table.insert(lines, "Combat Security Probe Report")
	table.insert(lines, "========================================")
	table.insert(lines, "Runner: " .. LocalPlayer.Name .. " (" .. tostring(LocalPlayer.UserId) .. ")")
	table.insert(lines, "Target: " .. (selectedPlayer and selectedPlayer.Name or "nil"))
	table.insert(lines, "PlaceId: " .. tostring(game.PlaceId))
	table.insert(lines, "JobId: " .. tostring(game.JobId))
	table.insert(lines, "Studio: " .. tostring(RunService:IsStudio()))
	table.insert(lines, "")
	table.insert(lines, string.format("Summary: %d vulnerable, %d blocked, %d skipped.", v, b, s))
	table.insert(lines, "")

	for i, item in ipairs(resultData) do
		table.insert(lines, string.format("%02d. [%s] %s", i, item.State, item.Name))
		if item.Detail then
			table.insert(lines, "    " .. tostring(item.Detail))
		end
	end

	return table.concat(lines, "\n")
end

local function copyReport()
	local report = buildReport()

	if typeof(setclipboard) == "function" then
		local ok = pcall(function()
			setclipboard(report)
		end)

		if ok then
			setStatus("Report copied", Color3.fromRGB(45, 105, 65))
			return
		end
	end

	print(report)
	setStatus("Clipboard unavailable; printed report", Color3.fromRGB(70, 80, 110))
end

local function refreshTarget()
	if not selectedPlayer then
		targetButton.Text = "Target: none"
		targetButton.BackgroundColor3 = Color3.fromRGB(120, 70, 45)
		return
	end

	if selectedPlayer == LocalPlayer then
		targetButton.Text = "Target: Me"
		targetButton.BackgroundColor3 = Color3.fromRGB(60, 120, 80)
		return
	end

	if isTargetAllowed(selectedPlayer) then
		targetButton.Text = "Target: " .. selectedPlayer.Name
		targetButton.BackgroundColor3 = Color3.fromRGB(60, 120, 80)
	else
		targetButton.Text = "Target: " .. selectedPlayer.Name
		targetButton.BackgroundColor3 = Color3.fromRGB(120, 70, 45)
	end
end

targetButton.MouseButton1Click:Connect(function()
	local list = Players:GetPlayers()

	table.sort(list, function(a, b)
		return a.Name < b.Name
	end)

	local index = table.find(list, selectedPlayer) or 1
	index += 1

	if index > #list then
		index = 1
	end

	selectedPlayer = list[index]
	refreshTarget()
end)

Players.PlayerAdded:Connect(refreshTarget)
Players.PlayerRemoving:Connect(function(player)
	if selectedPlayer == player then
		selectedPlayer = LocalPlayer
	end

	refreshTarget()
end)

local function precheck(testName)
	if not mainOk or not MainEvent then
		skipped(testName, "Remotes not loaded")
		return nil
	end

	local parts, err = getTargetParts()
	if not parts then
		skipped(testName, err)
		setStatus(err, Color3.fromRGB(120, 70, 45))
		return nil
	end

	return parts
end

local function testDamage()
	local name = "Damage"
	local p = precheck(name)
	if not p then return end

	setStatus("Testing damage...", Color3.fromRGB(45, 65, 100))

	local amount = n(damageBox.Text, DEFAULTS.Damage, 0, CAPS.MaxDamage)
	damageBox.Text = tostring(amount)

	if amount <= 0 then
		skipped(name, "Damage amount is 0")
		return
	end

	local before = p.Humanoid.Health

	fireServerFunction("Main_Damger_OC_R", {
		p.Character,
		{
			D = amount,
			F = LocalPlayer.Character or p.Character,
		},
	})

	local changed = waitUntil(getTimeout(), function()
		return p.Humanoid.Health < before - 0.01
	end)

	local after = p.Humanoid.Health

	if changed then
		vuln(name, string.format("Health %.2f -> %.2f", before, after))
	else
		blocked(name, "Health unchanged")
	end
end

local function testStateCombo()
	local name = "State x3"
	local p = precheck(name)
	if not p then return end

	setStatus("Testing Stun + CantHit + Ragdoll...", Color3.fromRGB(45, 65, 100))

	if p.Ragdoll.Value == true then
		skipped(name, "Target is already ragdolled")
		return
	end

	local stunValue = n(stunBox.Text, DEFAULTS.Stun, 0, CAPS.MaxStun)
	local cantHitValue = n(cantHitBox.Text, DEFAULTS.CantHit, 0, CAPS.MaxCantHit)
	local duration = n(ragTimeBox.Text, DEFAULTS.RagdollTime, 0, CAPS.MaxRagdollTime)
	local jumps = n(ragJumpBox.Text, DEFAULTS.RagdollJumps, 0, CAPS.MaxRagdollJumps)

	if p.Stun.Value == stunValue then
		stunValue = stunValue == 1 and 2 or 1
	end

	if p.CantHit.Value == cantHitValue then
		cantHitValue = math.clamp(cantHitValue + 1, 0, CAPS.MaxCantHit)
	end

	stunBox.Text = tostring(stunValue)
	cantHitBox.Text = tostring(cantHitValue)
	ragTimeBox.Text = tostring(duration)
	ragJumpBox.Text = tostring(jumps)

	local beforeStun = p.Stun.Value
	local beforeCantHit = p.CantHit.Value
	local beforeRagdoll = p.Ragdoll.Value

	fireServerFunction("ChangSTuNNNNNNN", {
		stunValue,
		p.Character,
	})

	fireServerFunction("ChangCanTHiTTT", {
		cantHitValue,
		p.Character,
	})

	fireServerFunction("Ragdoll_Character", {
		p.Character,
		true,
		{
			T = duration,
			J = jumps,
		},
	})

	waitUntil(getTimeout(), function()
		return p.Stun.Value == stunValue
			and p.CantHit.Value == cantHitValue
			and p.Ragdoll.Value == true
	end)

	local gotStun = p.Stun.Value == stunValue
	local gotCantHit = p.CantHit.Value == cantHitValue
	local gotRagdoll = p.Ragdoll.Value == true

	local accepted = {}

	if gotStun then
		table.insert(accepted, "Stun")
	end

	if gotCantHit then
		table.insert(accepted, "CantHit")
	end

	if gotRagdoll then
		table.insert(accepted, "Ragdoll")
	end

	if #accepted > 0 then
		vuln(
			name,
			string.format(
				"Accepted: %s | Stun %s->%s, CantHit %s->%s, Ragdoll %s->%s",
				table.concat(accepted, ", "),
				tostring(beforeStun),
				tostring(p.Stun.Value),
				tostring(beforeCantHit),
				tostring(p.CantHit.Value),
				tostring(beforeRagdoll),
				tostring(p.Ragdoll.Value)
			)
		)

		if gotStun then
			fireServerFunction("ChangSTuNNNNNNN", {
				0,
				p.Character,
			})
		end

		if gotCantHit then
			fireServerFunction("ChangCanTHiTTT", {
				0,
				p.Character,
			})
		end

		if gotRagdoll then
			task.wait(math.min(duration, 0.5))

			fireServerFunction("Ragdoll_Character", {
				p.Character,
				false,
			})
		end
	else
		blocked(name, "No Stun/CantHit/Ragdoll state changed")
	end
end

local function testALVC()
	local name = "ALVC"
	local p = precheck(name)
	if not p then return end

	setStatus("Testing ALVC...", Color3.fromRGB(45, 65, 100))

	local vx = n(velXBox.Text, DEFAULTS.VelX, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vy = n(velYBox.Text, DEFAULTS.VelY, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vz = n(velZBox.Text, DEFAULTS.VelZ, -CAPS.MaxVelocity, CAPS.MaxVelocity)

	velXBox.Text = tostring(vx)
	velYBox.Text = tostring(vy)
	velZBox.Text = tostring(vz)

	local vec = Vector3.new(vx, vy, vz)

	if vec.Magnitude > CAPS.MaxVelocity then
		vec = vec.Unit * CAPS.MaxVelocity
	end

	local before = p.HRP.AssemblyLinearVelocity

	fireServerFunction("ALVC", {
		p.HRP,
		vec,
		{
			C = p.Character,
		},
	})

	local changed = waitUntil(getTimeout(), function()
		return (p.HRP.AssemblyLinearVelocity - before).Magnitude > 4
	end)

	if changed then
		vuln(name, "Velocity changed after client request")
	else
		blocked(name, "Velocity unchanged")
	end
end
