local function n(value, fallback, minValue, maxValue)
	local number = tonumber(value)

	if number == nil then
		number = fallback
	end

	if minValue ~= nil then
		number = math.max(minValue, number)
	end

	if maxValue ~= nil then
		number = math.min(maxValue, number)
	end

	return number
end

local function clampVector(vec, maxMagnitude)
	if vec.Magnitude > maxMagnitude then
		return vec.Unit * maxMagnitude
	end

	return vec
end

local function waitUntil(timeout, predicate)
	local start = os.clock()

	while os.clock() - start < timeout do
		local ok, result = pcall(predicate)

		if ok and result then
			return true
		end

		task.wait(0.04)
	end

	return false
end

local function isTargetAllowed(player)
	if not REQUIRE_TARGET_CONSENT then
		return true
	end

	if not player then
		return false
	end

	if player == LocalPlayer then
		return true
	end

	if player:GetAttribute(CONSENT_ATTRIBUTE) == true then
		return true
	end

	local character = player.Character
	if character then
		if character:GetAttribute(CONSENT_ATTRIBUTE) == true then
			return true
		end

		if character:GetAttribute(TARGET_ATTRIBUTE) == true then
			return true
		end
	end

	return false
end

local function getTargetParts()
	if not selectedPlayer then
		return nil, "No selected player"
	end

	if not isTargetAllowed(selectedPlayer) then
		return nil, selectedPlayer.Name .. " has not opted in"
	end

	local character = selectedPlayer.Character
	if not character then
		return nil, "Target has no character"
	end

	local humanoid = character:FindFirstChild("Humanoid")
	local hrp = character:FindFirstChild("HumanoidRootPart")
	local stats = character:FindFirstChild("CharcterStats")

	if not humanoid then
		return nil, "Target missing Humanoid"
	end

	if not hrp then
		return nil, "Target missing HumanoidRootPart"
	end

	if not stats then
		return nil, "Target missing CharcterStats"
	end

	return {
		Character = character,
		Humanoid = humanoid,
		HRP = hrp,
		Stats = stats,
	}, nil
end

local function fireServerFunction(functionName, args)
	if not MainEvent then
		return false, "MainEvent missing"
	end

	MainEvent:FireServer(REMOTE_CODE, functionName, args)

	return true
end

local function getTimeout()
	return n(DEFAULTS.Timeout, DEFAULTS.Timeout, CAPS.MinTimeout, CAPS.MaxTimeout)
end

--// Draggable top bar
do
	local dragging = false
	local dragStart
	local startPos

	top.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not dragging then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local delta = input.Position - dragStart

		frame.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end)
end

local function refreshTarget()
	if not selectedPlayer then
		targetButton.Text = "Target: none"
		targetButton.BackgroundColor3 = Color3.fromRGB(120, 70, 45)
		return
	end

	if selectedPlayer == LocalPlayer then
		targetButton.Text = "Target: Me"
		targetButton.BackgroundColor3 = Color3.fromRGB(60, 120, 80)
		return
	end

	if isTargetAllowed(selectedPlayer) then
		targetButton.Text = "Target: " .. selectedPlayer.Name
		targetButton.BackgroundColor3 = Color3.fromRGB(60, 120, 80)
	else
		targetButton.Text = selectedPlayer.Name .. " (No)"
		targetButton.BackgroundColor3 = Color3.fromRGB(120, 70, 45)
	end
end

targetButton.MouseButton1Click:Connect(function()
	local list = Players:GetPlayers()

	table.sort(list, function(a, b)
		return a.Name < b.Name
	end)

	local index = table.find(list, selectedPlayer) or 1
	index += 1

	if index > #list then
		index = 1
	end

	selectedPlayer = list[index]
	refreshTarget()
end)

Players.PlayerAdded:Connect(refreshTarget)

Players.PlayerRemoving:Connect(function(player)
	if selectedPlayer == player then
		selectedPlayer = LocalPlayer
	end

	refreshTarget()
end)

refreshTarget()

local function precheck(testName)
	if not mainOk or not MainEvent then
		setStatus("Remotes not loaded", Color3.fromRGB(120, 70, 45))
		return nil
	end

	local parts, err = getTargetParts()
	if not parts then
		setStatus(err, Color3.fromRGB(120, 70, 45))
		return nil
	end

	return parts
end

--// Individual tests
--// ----------------------------------------------------

local function testDamage()
	local p = precheck("Damage")
	if not p then return end

	local amount = n(damageBox.Text, DEFAULTS.Damage, 0, CAPS.MaxDamage)
	damageBox.Text = tostring(amount)

	if amount <= 0 then
		setStatus("Damage amount is 0", Color3.fromRGB(120, 70, 45))
		return
	end

	local before = p.Humanoid.Health

	fireServerFunction("Main_Damger_OC_R", {
		p.Character,
		{
			D = amount,
			F = LocalPlayer.Character or p.Character,
		},
	})

	local changed = waitUntil(getTimeout(), function()
		return p.Humanoid.Health < before - 0.01
	end)

	local after = p.Humanoid.Health

	if changed then
		setStatus(string.format("Damage accepted %.2f -> %.2f", before, after), Color3.fromRGB(120, 45, 45))
	else
		setStatus("Damage blocked", Color3.fromRGB(45, 105, 65))
	end
end

local function testALVC()
	local p = precheck("ALVC")
	if not p then return end

	local vx = n(velXBox.Text, DEFAULTS.VelX, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vy = n(velYBox.Text, DEFAULTS.VelY, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vz = n(velZBox.Text, DEFAULTS.VelZ, -CAPS.MaxVelocity, CAPS.MaxVelocity)

	velXBox.Text = tostring(vx)
	velYBox.Text = tostring(vy)
	velZBox.Text = tostring(vz)

	local vec = Vector3.new(vx, vy, vz)

	if vec.Magnitude > CAPS.MaxVelocity then
		vec = vec.Unit * CAPS.MaxVelocity
	end

	local before = p.HRP.AssemblyLinearVelocity

	fireServerFunction("ALVC", {
		p.HRP,
		vec,
		{
			C = p.Character,
		},
	})

	local changed = waitUntil(getTimeout(), function()
		return (p.HRP.AssemblyLinearVelocity - before).Magnitude > 4
	end)

	if changed then
		setStatus("Force accepted", Color3.fromRGB(120, 45, 45))
	else
		setStatus("Force blocked", Color3.fromRGB(45, 105, 65))
	end
end
