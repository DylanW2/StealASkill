local function tween(object, info, props)
	local t = TweenService:Create(object, info, props)
	t:Play()
	return t
end

local function addCorner(parent, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius or 8)
	corner.Parent = parent
	return corner
end

local function addStroke(parent, color, transparency, thickness)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color or Color3.fromRGB(255, 255, 255)
	stroke.Transparency = transparency or 0.88
	stroke.Thickness = thickness or 1
	stroke.Parent = parent
	return stroke
end

local function getBrightness(color)
	return (color.R * 0.299) + (color.G * 0.587) + (color.B * 0.114)
end

local function getReadableTextColor(bg)
	if getBrightness(bg) > 0.55 then
		return Color3.fromRGB(18, 18, 24)
	end

	return Color3.fromRGB(255, 255, 255)
end

local function escapeRichText(text)
	text = tostring(text or "")

	text = text:gsub("&", "&amp;")
	text = text:gsub("<", "&lt;")
	text = text:gsub(">", "&gt;")
	text = text:gsub("\"", "&quot;")
	text = text:gsub("'", "&apos;")

	return text
end

local function colorToHex(color)
	return string.format(
		"#%02X%02X%02X",
		math.floor(color.R * 255 + 0.5),
		math.floor(color.G * 255 + 0.5),
		math.floor(color.B * 255 + 0.5)
	)
end

local function trim(s)
	s = tostring(s or "")
	return s:match("^%s*(.-)%s*$")
end

local function lighten(color, amount)
	return color:Lerp(Color3.fromRGB(255, 255, 255), amount or 0.08)
end

local buttonBaseColors = {}
local buttonHovering = {}

local function setButtonBase(button, color, textColor)
	buttonBaseColors[button] = color
	button.BackgroundColor3 = color

	if textColor then
		button.TextColor3 = textColor
	end
end

local function applyButtonVisual(button)
	local base = buttonBaseColors[button] or button.BackgroundColor3

	if buttonHovering[button] then
		tween(button, TWEEN_FAST, {
			BackgroundColor3 = lighten(base, 0.08),
		})
	else
		tween(button, TWEEN_FAST, {
			BackgroundColor3 = base,
		})
	end
end

local function makeButton(name, text, pos, size, bgColor, textColor)
	local btn = Instance.new("TextButton")
	btn.Name = name
	btn.Size = size
	btn.Position = pos
	btn.BackgroundColor3 = bgColor or THEME.Button
	btn.BorderSizePixel = 0
	btn.Text = text
	btn.TextColor3 = textColor or THEME.Text
	btn.TextSize = 13
	btn.Font = Enum.Font.GothamBold
	btn.AutoButtonColor = false
	btn.TextTruncate = Enum.TextTruncate.AtEnd
	btn.ZIndex = 4
	btn.Parent = ContentFrame

	addCorner(btn, 8)
	addStroke(btn, Color3.fromRGB(255, 255, 255), 0.93, 1)

	buttonBaseColors[btn] = bgColor or THEME.Button
	buttonHovering[btn] = false

	-- No size/position press tween.
	-- That was the "shake" when flipping colors/fonts.
	btn.MouseEnter:Connect(function()
		buttonHovering[btn] = true
		applyButtonVisual(btn)
	end)

	btn.MouseLeave:Connect(function()
		buttonHovering[btn] = false
		applyButtonVisual(btn)
	end)

	btn.MouseButton1Down:Connect(function()
		local base = buttonBaseColors[btn] or btn.BackgroundColor3
		tween(btn, TWEEN_FAST, {
			BackgroundColor3 = base:Lerp(Color3.fromRGB(0, 0, 0), 0.08),
		})
	end)

	btn.MouseButton1Up:Connect(function()
		applyButtonVisual(btn)
	end)

	return btn
end

local function updatePreview()
	local raw = trim(MessageBox.Text)

	if raw == "" then
		PreviewText.Text = "Your styled message appears here"
		PreviewText.TextColor3 = selectedColor
		PreviewText.Font = selectedFont
		PreviewStroke.Transparency = 1
		return
	end

	PreviewText.Text = escapeRichText(raw)
	PreviewText.TextColor3 = selectedColor
	PreviewText.Font = selectedFont
	PreviewStroke.Transparency = selectedStrokeEnabled and 0.2 or 1
end

local function updateColorButton()

	selectedColor = presetData.color

	ColorButton.Text = presetData.name
	ColorButton.TextColor3 = getReadableTextColor(selectedColor)
	setButtonBase(ColorButton, selectedColor, getReadableTextColor(selectedColor))

	PreviewText.TextColor3 = selectedColor

local function updateFontButton()

	selectedFont = presetData.font

	FontButton.Font = selectedFont
	MessageBox.Font = selectedFont
	PreviewText.Font = selectedFont

	updatePreview()
end

		setButtonBase(BelowButton, THEME.Button, THEME.SubText)

		setButtonBase(AboveButton, THEME.Button, THEME.SubText)

local function updateStrokeButton()

		StrokeButton.Text = "Stroke: On"
		setButtonBase(StrokeButton, THEME.ButtonActive, THEME.Text)
	else
		StrokeButton.Text = "Stroke: Off"
		setButtonBase(StrokeButton, THEME.Button, THEME.SubText)
	end

	applyButtonVisual(StrokeButton)
	updatePreview()
end

local function getFormattedMessage(rawText)
	local clean = escapeRichText(rawText)
	local hex = colorToHex(selectedColor)

	-- THIS is the actual sent font change.
	-- The previous script only changed the GUI preview font.
	-- This one sends face="" in RichText.
	local formatted = string.format(
		"<font color=\"%s\" face=\"%s\">%s</font>",
		hex,
		selectedFontFace,
		clean
	)

	if selectedStrokeEnabled then
		formatted = "<stroke color=\"#000000\" thickness=\"1.5\" transparency=\"0.2\">" .. formatted .. "</stroke>"
	end

	return formatted
end

local function sendMessageToServer(formattedMessage, rawMessage, options)
	fireServerFunction("Send_Above_Text", {
		formattedMessage,
		"AC",
		{
			AHT = options.Above,

			-- Extra metadata. If the server ignores it, harmless.
			Font = options.FontFace,
			FontName = options.FontName,
			Color = options.ColorHex,
			Stroke = options.Stroke,
		}
	})

	return true
end

local sending = false

local function sendCurrentMessage()
	if sending then
		return
	end

	local raw = trim(MessageBox.Text)

	if raw == "" then
		setStatus("✗ Enter a message", THEME.Red)

		tween(MessageBox, TWEEN_FAST, {
			BackgroundColor3 = Color3.fromRGB(58, 35, 42),
		})

		task.delay(0.18, function()
			if MessageBox and MessageBox.Parent then
				tween(MessageBox, TWEEN_FAST, {
					BackgroundColor3 = THEME.Card,
				})
			end
		end)

		return
	end

	sending = true

	local formatted = getFormattedMessage(raw)
	local fontName = fontPresets[fontIndex].name

	local ok, result = pcall(function()
		return sendMessageToServer(formatted, raw, {
			Above = isAbove,
			Font = selectedFont,
			FontFace = selectedFontFace,
			FontName = fontName,
			Color = selectedColor,
			ColorHex = colorToHex(selectedColor),
			Stroke = selectedStrokeEnabled,
		})
	end)

	if not ok or result == false then
		setStatus("✗ Failed to send", THEME.Red)
		warn("Send failed:", result)
		sending = false
		return
	end

	setStatus("✓ Sent", Color3.fromRGB(110, 255, 140))

	local originalText = SendButton.Text
	SendButton.Text = "Sent"
	setButtonBase(SendButton, Color3.fromRGB(95, 240, 155), Color3.fromRGB(255, 255, 255))
	applyButtonVisual(SendButton)

	task.delay(0.30, function()
		if SendButton and SendButton.Parent then
			SendButton.Text = originalText
			setButtonBase(SendButton, THEME.Green, Color3.fromRGB(255, 255, 255))
			applyButtonVisual(SendButton)
		end

		sending = false
	end)
end

--// ============================================================
--// CUSTOM DRAGGING
--// ============================================================

local dragging = false
local dragStart = nil
local startPos = nil
local shadowStartPos = nil

local function updateDrag(input)
	local delta = input.Position - dragStart

	local newPos = UDim2.new(
		startPos.X.Scale,
		startPos.X.Offset + delta.X,
		startPos.Y.Scale,
		startPos.Y.Offset + delta.Y
	)

	MainFrame.Position = newPos

	Shadow.Position = UDim2.new(
		shadowStartPos.X.Scale,
		shadowStartPos.X.Offset + delta.X,
		shadowStartPos.Y.Scale,
		shadowStartPos.Y.Offset + delta.Y
	)
end

HeaderFrame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = MainFrame.Position
		shadowStartPos = Shadow.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		updateDrag(input)
	end
end)
