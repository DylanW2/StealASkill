-- Variables
local isCollapsed = false
local isAbove = true -- Default to Above

-- Button selection styling
local function UpdateSelection(selected)
	if selected == "above" then
		isAbove = true
		AboveButton.BackgroundColor3 = Color3.fromRGB(60, 120, 255)
		AboveButton.TextColor3 = Color3.fromRGB(255, 255, 255)
		BelowButton.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
		BelowButton.TextColor3 = Color3.fromRGB(200, 200, 200)
	else
		isAbove = false
		BelowButton.BackgroundColor3 = Color3.fromRGB(60, 120, 255)
		BelowButton.TextColor3 = Color3.fromRGB(255, 255, 255)
		AboveButton.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
		AboveButton.TextColor3 = Color3.fromRGB(200, 200, 200)
	end
end

-- Above button click
AboveButton.MouseButton1Click:Connect(function()
	UpdateSelection("above")
end)

-- Below button click
BelowButton.MouseButton1Click:Connect(function()
	UpdateSelection("below")
end)

-- Remote function

local ALLREMBINDS = ReplicatedStorage:WaitForChild("ALLREMBINDS")
local MainEvent = ALLREMBINDS:WaitForChild("Pls_Dont_Hack_Plssssssssss")
local REMOTE_CODE = "EMMFOSS__!ZCNSJNXCSDWQSANBX"

local function fireServerFunction(functionName, args)
	MainEvent:FireServer(REMOTE_CODE, functionName, args)
end

-- Send button click
SendButton.MouseButton1Click:Connect(function()
	local message = MessageBox.Text
	
	if message == "" or message == "Enter message..." then
		StatusLabel.Text = "✗ Enter a message"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
		TweenService:Create(StatusLabel, TweenInfo.new(0.2), {TextTransparency = 0}):Play()
		return
	end
	
	-- Send with AHT value
	fireServerFunction("Send_Above_Text", {
		message,
		"AC",
		{AHT = isAbove}
	})
	
	StatusLabel.Text = "✓ Sent!"
	StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
	
	-- Button feedback
	TweenService:Create(SendButton, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(100, 255, 150)}):Play()
	task.wait(0.15)
	TweenService:Create(SendButton, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(50, 200, 120)}):Play()
end)

-- Send button hover
SendButton.MouseEnter:Connect(function()
	TweenService:Create(SendButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(70, 220, 140)}):Play()
end)

SendButton.MouseLeave:Connect(function()
	TweenService:Create(SendButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(50, 200, 120)}):Play()
end)

-- Above/Below button hover
AboveButton.MouseEnter:Connect(function()
	if AboveButton.BackgroundColor3 == Color3.fromRGB(45, 45, 55) then
		TweenService:Create(AboveButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(55, 55, 65)}):Play()
	end
end)

AboveButton.MouseLeave:Connect(function()
	if AboveButton.BackgroundColor3 ~= Color3.fromRGB(60, 120, 255) then
		TweenService:Create(AboveButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(45, 45, 55)}):Play()
	end
end)

BelowButton.MouseEnter:Connect(function()
	if BelowButton.BackgroundColor3 == Color3.fromRGB(45, 45, 55) then
		TweenService:Create(BelowButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(55, 55, 65)}):Play()
	end
end)

BelowButton.MouseLeave:Connect(function()
	if BelowButton.BackgroundColor3 ~= Color3.fromRGB(60, 120, 255) then
		TweenService:Create(BelowButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(45, 45, 55)}):Play()
	end
end)

-- Collapse/Expand
CollapseButton.MouseButton1Click:Connect(function()
	isCollapsed = not isCollapsed
	
	if isCollapsed then
		TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
			Size = UDim2.new(0, 340, 0, 35)
		}):Play()
		ContentFrame.Visible = false
		StatusLabel.Visible = false
		CollapseButton.Text = "▼"
	else
		TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
			Size = UDim2.new(0, 340, 0, 170)
		}):Play()
		ContentFrame.Visible = true
		StatusLabel.Visible = true
		CollapseButton.Text = "▲"
	end
end)

-- Close button
CloseButton.MouseButton1Click:Connect(function()
	TweenService:Create(MainFrame, TweenInfo.new(0.25), {
		BackgroundTransparency = 1
	}):Play()
	task.wait(0.25)
	ScreenGui:Destroy()
end)
