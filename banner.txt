local function fireServerFunction(functionName, args)
	return MainEvent:InvokeServer(REMOTE_CODE, functionName, args)
end

local function setPlayerRebirths(targetPlayer)
	if not targetPlayer then
		return false, "No target player"
	end

	local success, result = pcall(function()
		return fireServerFunction("FindObject_ExecuteOnIt", {
			targetPlayer,
			{"leaderstats", "Rebirths"},
			{
				F = "ChangeProptyOfSmtHSHHH__",
				V = {
					"__OBjCtaAA",
					"Value",
					REBIRTH_VALUE,
					{FP = targetPlayer}
				},
				FOBO = true
			}
		})
	end)

	if success then
		return true, result
	else
		return false, result
	end
end

local function New(className, props, parent)
	local obj = Instance.new(className)

	for k, v in pairs(props or {}) do
		obj[k] = v
	end

	if parent then
		obj.Parent = parent
	end

	return obj
end

local function corner(parent, radius)
	return New("UICorner", {
		CornerRadius = UDim.new(0, radius),
	}, parent)
end

local function stroke(parent, color, thickness, transparency)
	return New("UIStroke", {
		Color = color,
		Thickness = thickness or 1,
		Transparency = transparency or 0,
	}, parent)
end

local function padding(parent, l, r, t, b)
	return New("UIPadding", {
		PaddingLeft = UDim.new(0, l or 0),
		PaddingRight = UDim.new(0, r or 0),
		PaddingTop = UDim.new(0, t or 0),
		PaddingBottom = UDim.new(0, b or 0),
	}, parent)
end

local function clearList(frame)
	for _, child in ipairs(frame:GetChildren()) do
		if child:IsA("TextButton") or child:IsA("TextLabel") then
			child:Destroy()
		end
	end
end

local function setDropdown(open)
	dropdownOpen = open
	Dropdown.Visible = open

	if selectedPlayer then
		PlayerButton.Text = selectedPlayer.Name .. (open and " ▲" or " ▼")
	else
		PlayerButton.Text = "Select player..." .. (open and " ▲" or " ▼")
	end
end

local function updateDropdown()
	clearList(Dropdown)

	local playerList = Players:GetPlayers()

	table.sort(playerList, function(a, b)
		return string.lower(a.Name) < string.lower(b.Name)
	end)

	for _, player in ipairs(playerList) do
		local button = New("TextButton", {
			Size = UDim2.new(1, -4, 0, 30),
			BackgroundColor3 = player == selectedPlayer and Color3.fromRGB(65, 75, 100) or Color3.fromRGB(45, 45, 54),
			BorderSizePixel = 0,
			AutoButtonColor = false,
			Font = Enum.Font.Gotham,
			TextSize = 13,
			TextColor3 = Color3.fromRGB(240, 240, 245),
			TextXAlignment = Enum.TextXAlignment.Left,
			Text = "  " .. player.Name,
			ZIndex = 21,
		}, Dropdown)

		corner(button, 6)

		button.MouseEnter:Connect(function()
			button.BackgroundColor3 = Color3.fromRGB(58, 58, 70)
		end)

		button.MouseLeave:Connect(function()
			if player == selectedPlayer then
				button.BackgroundColor3 = Color3.fromRGB(65, 75, 100)
			else
				button.BackgroundColor3 = Color3.fromRGB(45, 45, 54)
			end
		end)

		button.MouseButton1Click:Connect(function()
			selectedPlayer = player
			SelectedLabel.Text = "Selected: " .. player.Name
			StatusLabel.Text = "Ready"
			setDropdown(false)
			updateDropdown()
		end)
	end
end

PlayerButton.MouseButton1Click:Connect(function()

end)

SetButton.MouseButton1Click:Connect(function()
	if not selectedPlayer then
		StatusLabel.Text = "Select a player first"
		return
	end

	if not selectedPlayer.Parent then
		StatusLabel.Text = "Selected player left"
		selectedPlayer = nil
		SelectedLabel.Text = "No player selected"
		setDropdown(false)
		updateDropdown()
		return
	end

	SetButton.Text = "Setting..."
	SetButton.BackgroundColor3 = Color3.fromRGB(95, 95, 95)
	StatusLabel.Text = "Banning " .. selectedPlayer.Name .. "..."

	local success, result = setPlayerRebirths(selectedPlayer)

	if success then
		StatusLabel.Text = "Set rebirths for " .. selectedPlayer.Name
	else
		StatusLabel.Text = "Failed: " .. tostring(result)
		warn("Failed to set rebirths:", result)
	end

	SetButton.Text = "Set Rebirths"
	SetButton.BackgroundColor3 = Color3.fromRGB(55, 90, 60)
end)

Players.PlayerRemoving:Connect(function(player)
	if selectedPlayer == player then
		selectedPlayer = nil
		SelectedLabel.Text = "No player selected"
		StatusLabel.Text = "Selected player left"
		setDropdown(false)
	end

	updateDropdown()
end)

do
	local dragging = false
	local dragStart
	local startPosition

	TitleBar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStart = input.Position
			startPosition = Main.Position
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			local delta = input.Position - dragStart

			Main.Position = UDim2.new(
				startPosition.X.Scale,
				startPosition.X.Offset + delta.X,
				startPosition.Y.Scale,
				startPosition.Y.Offset + delta.Y
			)
		end
	end)
end
