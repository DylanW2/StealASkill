local function isWhitelistedMagic(magicModelOrName)
	if typeof(magicModelOrName) == "Instance" then
		return DO_NOT_SELL[magicModelOrName.Name] == true
	end

	return DO_NOT_SELL[tostring(magicModelOrName)] == true
end

local function getMagicAttribute(model)
	for _, attrName in ipairs(ATTRIBUTE_NAMES) do
		local value = model:GetAttribute(attrName)

		if value ~= nil then
			return attrName, tostring(value)
		end
	end

	return nil, nil
end

local function magicBelongsToPlayer(player, magicModel)
	if not player or not magicModel then
		return false
	end

	local _, value = getMagicAttribute(magicModel)

	if not value then
		return false
	end

	local userId = tostring(player.UserId)
	local name = player.Name

	return string.find(value, userId, 1, true) ~= nil
		or string.find(value, name, 1, true) ~= nil
end

local function getPlayerMagics(player)
	local results = {}

	if not player then
		return results
	end

	for _, child in ipairs(MagicsIR:GetChildren()) do
		if child:IsA("Model") and magicBelongsToPlayer(player, child) then
			local attrName, attrValue = getMagicAttribute(child)

			table.insert(results, {
				Model = child,
				Name = child.Name,
				AttributeName = attrName or "?",
				AttributeValue = attrValue or "?",
			})
		end
	end

	table.sort(results, function(a, b)
		return string.lower(a.Name) < string.lower(b.Name)
	end)

	return results
end

local function sortMagicsForStealing(magics)
	table.sort(magics, function(a, b)
		local aPriority = STEAL_MAGIC_PRIORITY[a.Name] or math.huge
		local bPriority = STEAL_MAGIC_PRIORITY[b.Name] or math.huge

		if aPriority == bPriority then
			return string.lower(a.Name) < string.lower(b.Name)
		end

		return aPriority < bPriority
	end)
end

local function waitForLocalMagicByName(magicName, timeout)
	local started = os.clock()

	while os.clock() - started < timeout do
		local localMagics = getPlayerMagics(LocalPlayer)

		for _, info in ipairs(localMagics) do
			if info.Name == magicName and info.Model and info.Model.Parent then
				return info.Model
			end
		end

		task.wait()
	end

	return nil
end

local function getPromptFromMagic(magicModel, promptName)
	if not magicModel or not magicModel.Parent then
		return nil
	end

	local hrp = magicModel:FindFirstChild("HumanoidRootPart")
	if not hrp then
		hrp = magicModel:FindFirstChild("HumanoidRootPart", true)
	end

	if not hrp then
		return nil
	end

	local prompt = hrp:FindFirstChild(promptName)

	if prompt and prompt:IsA("ProximityPrompt") then
		return prompt
	end

	return nil
end

local function getPlayerBase(player)
	if not player then
		return nil
	end

	local userId = tostring(player.UserId)

	for _, base in ipairs(Bases:GetChildren()) do
		local owner = base:GetAttribute("Owner")

		if owner ~= nil then
			local ownerText = tostring(owner)

			if string.find(ownerText, userId, 1, true) then
				return base
			end
		end
	end

	return nil
end

local function getBaseCollectZoneCFrame(base)
	if not base then
		return nil
	end

	local collectZoneFolder = base:FindFirstChild(COLLECT_ZONE_FOLDER_NAME)
	if not collectZoneFolder then
		return nil
	end

	local collectZone = collectZoneFolder:FindFirstChild(COLLECT_ZONE_PART_NAME)
	if not collectZone then
		return nil
	end

	if collectZone:IsA("BasePart") then
		return collectZone.CFrame * CFrame.new(0, 3, 0)
	end

	if collectZone:IsA("Model") then
		return collectZone:GetPivot() * CFrame.new(0, 3, 0)
	end

	return nil
end

local function teleportLocalPlayerToCFrame(cf)
	local root = getCharacterRoot()

	if not root or not cf then
		return false
	end

	root.AssemblyLinearVelocity = Vector3.zero
	root.AssemblyAngularVelocity = Vector3.zero
	root.CFrame = cf

	task.wait(TELEPORT_SETTLE_TIME)

	return true
end

local function returnToLocalPlayerBase()
	local base = getPlayerBase(LocalPlayer)

	if not base then
		return false, "local player base not found"
	end

	local cf = getBaseCollectZoneCFrame(base)

	if not cf then
		return false, "collect zone not found at " .. base.Name .. "." .. COLLECT_ZONE_FOLDER_NAME .. "." .. COLLECT_ZONE_PART_NAME
	end

	local success = teleportLocalPlayerToCFrame(cf)

	if success then
		task.wait(BASE_SETTLE_TIME)
		return true, "returned to collect zone"
	end

	return false, "failed to return"
end

local function teleportToPrompt(prompt, customSettleTime)
	local root = getCharacterRoot()

	if not root then
		return false
	end

	local promptPart = prompt.Parent

	if not promptPart or not promptPart:IsA("BasePart") then
		return false
	end

	local maxDistance = prompt.MaxActivationDistance
	if maxDistance <= 0 then
		maxDistance = 10
	end

	local offsetDistance = math.clamp(maxDistance * 0.35, 1.5, 5)
	local targetCFrame = promptPart.CFrame * CFrame.new(0, 0, offsetDistance)

	root.AssemblyLinearVelocity = Vector3.zero
	root.AssemblyAngularVelocity = Vector3.zero
	root.CFrame = targetCFrame

	task.wait(customSettleTime or TELEPORT_SETTLE_TIME)

	return true
end

local function activatePrompt(prompt)
	if not prompt or not prompt.Parent then
		return false
	end

	if not prompt:IsA("ProximityPrompt") then
		return false
	end

	if not prompt.Enabled then
		return false
	end

	local fired = false
	local connection

	connection = prompt.Triggered:Connect(function(playerWhoTriggered)
		if playerWhoTriggered == nil or playerWhoTriggered == LocalPlayer then
			fired = true
		end
	end)

	prompt:InputHoldBegin()

	local holdTime = tonumber(prompt.HoldDuration) or 0
	task.wait(math.max(holdTime + PROMPT_EXTRA_HOLD_TIME, PROMPT_MIN_HOLD_TIME))

	prompt:InputHoldEnd()

	local startedWaiting = os.clock()

	while not fired and prompt.Parent and os.clock() - startedWaiting < PROMPT_TIMEOUT do
		task.wait()
	end

	if connection then
		connection:Disconnect()
	end

	return fired
end

local function findLocalMagicToSellByName(magicName, originalModel)
	if originalModel and originalModel.Parent and not isWhitelistedMagic(originalModel) then
		local prompt = getSellPrompt(originalModel)

		if prompt then
			return originalModel
		end
	end

	local localMagics = getPlayerMagics(LocalPlayer)

	for _, info in ipairs(localMagics) do
		if info.Name == magicName and info.Model and info.Model.Parent and not isWhitelistedMagic(info.Model) then
			local prompt = getSellPrompt(info.Model)

			if prompt then
				return info.Model
			end
		end
	end

	return nil
end

updateResults = function()
	clearList(Results)

	if not selectedPlayer then
		StatsLabel.Text = "Select a player to load magics"
		emptyResult("No player selected")
		return
	end

	local magics = getPlayerMagics(selectedPlayer)
	local query = string.lower(SearchBox.Text or "")
	local shown = 0

	for _, info in ipairs(magics) do
		local nameLower = string.lower(info.Name)
		local attrLower = string.lower(info.AttributeValue)

		local matches = query == ""
			or string.find(nameLower, query, 1, true)
			or string.find(attrLower, query, 1, true)

		if matches then
			shown += 1

			local whitelisted = isWhitelistedMagic(info.Name)

			local button = New("TextButton", {
				Size = UDim2.new(1, -4, 0, 48),
				BackgroundColor3 = whitelisted and Color3.fromRGB(45, 85, 55) or Color3.fromRGB(45, 45, 54),
				BorderSizePixel = 0,
				AutoButtonColor = false,
				Font = Enum.Font.Gotham,
				TextSize = 13,
				TextColor3 = Color3.fromRGB(240, 240, 245),
				TextXAlignment = Enum.TextXAlignment.Left,
				TextYAlignment = Enum.TextYAlignment.Center,
				Text = "  " .. info.Name .. (whitelisted and " [WHITELISTED]" or "") .. "\n  " .. info.AttributeName .. ": " .. info.AttributeValue,
			}, Results)

			corner(button, 7)

			button.MouseEnter:Connect(function()
				button.BackgroundColor3 = Color3.fromRGB(58, 58, 70)
			end)

			button.MouseLeave:Connect(function()
				if isWhitelistedMagic(info.Name) then
					button.BackgroundColor3 = Color3.fromRGB(45, 85, 55)
				else
					button.BackgroundColor3 = Color3.fromRGB(45, 45, 54)
				end
			end)

			button.MouseButton1Click:Connect(function()
				print("Selected magic:", info.Name)
				print("Attribute:", info.AttributeName, info.AttributeValue)
				print("Path:", info.Model:GetFullName())
			end)
		end
	end

	StatsLabel.Text = string.format("Showing %d / %d magics for %s", shown, #magics, selectedPlayer.Name)

	if shown == 0 then
		emptyResult("No matching magics found")
	end
end

local function sellOneMagic(magicModel)
	if not magicModel or not magicModel.Parent then
		return false, "missing magic"
	end

	if isWhitelistedMagic(magicModel) then
		return false, "whitelisted"
	end

	local prompt = getSellPrompt(magicModel)

	if not prompt then
		return false, "missing sell prompt"
	end

	prompt.HoldDuration = 0.1

	if not teleportToPrompt(prompt) then
		return false, "teleport failed"
	end

	if activatePrompt(prompt) then
		return true, "sold"
	end

	return false, "prompt not fired"
end

local function sellAllLocalPlayerMagics()
	if sellingAll then
		return
	end

	sellingAll = true
	SellAllButton.Text = "Selling..."
	SellAllButton.BackgroundColor3 = Color3.fromRGB(95, 95, 95)

	local originalCFrame
	local root = getCharacterRoot()

	if root then
		originalCFrame = root.CFrame
	end

	local magics = getPlayerMagics(LocalPlayer)
	local total = #magics
	local sold = 0
	local failed = 0
	local skipped = 0

	if total == 0 then
		StatsLabel.Text = "No local-player magics found to sell"
	else
		for index, info in ipairs(magics) do
			if not sellingAll then
				break
			end

			if info.Model and info.Model.Parent then
				if isWhitelistedMagic(info.Name) then
					skipped += 1
					StatsLabel.Text = string.format("Skipping whitelisted %d / %d: %s", index, total, info.Name)
					task.wait(BETWEEN_ACTION_DELAY)
				else
					StatsLabel.Text = string.format("Selling local magic %d / %d: %s", index, total, info.Name)

					local success, reason = sellOneMagic(info.Model)

					if success then
						sold += 1
					else
						failed += 1
						warn("Failed to sell:", info.Name, reason)
					end

					task.wait(BETWEEN_MAGIC_DELAY)
				end
			end
		end

		StatsLabel.Text = string.format("Sell All done. Sold: %d | Failed: %d | Skipped: %d", sold, failed, skipped)
	end

	if originalCFrame then
		local newRoot = getCharacterRoot()

		if newRoot then
			newRoot.AssemblyLinearVelocity = Vector3.zero
			newRoot.AssemblyAngularVelocity = Vector3.zero
			newRoot.CFrame = originalCFrame
		end
	end

	SellAllButton.Text = "Sell All"
	SellAllButton.BackgroundColor3 = Color3.fromRGB(120, 65, 45)
	sellingAll = false

	task.wait(BETWEEN_ACTION_DELAY)
	updateResults()
end

local function stealOneMagic(magicModel)
	if not magicModel or not magicModel.Parent then
		return false, "missing magic"
	end

	local prompt = getStealPrompt(magicModel)

	if not prompt then
		return false, "missing steal prompt"
	end

	prompt.Enabled = true
	prompt.HoldDuration = 0.1

	if not teleportToPrompt(prompt, STEAL_TELEPORT_SETTLE_TIME) then
		return false, "teleport failed"
	end

	local fired = activatePrompt(prompt)

	task.wait(STEAL_MIN_STAY_AFTER_FIRE)

	if fired then
		return true, "steal prompt fired"
	end

	return false, "prompt not fired"
end

local function stealAllSelectedPlayerMagics()
	if stealingAll then
		return
	end

	if not selectedPlayer then
		StatsLabel.Text = "Select another player first"
		return
	end

	if selectedPlayer == LocalPlayer then
		StatsLabel.Text = "Steal All only works on other players"
		return
	end

	stealingAll = true
	updateStealAllButtonState()

	local magics = getPlayerMagics(selectedPlayer)
	sortMagicsForStealing(magics)

	local total = #magics
	local stolen = 0
	local stealFailed = 0
	local soldAfterSteal = 0
	local sellFailedAfterSteal = 0
	local skippedSelling = 0
	local returnedToCollectZone = 0

	if total == 0 then
		StatsLabel.Text = "No magics found for " .. selectedPlayer.Name
	else
		for index, info in ipairs(magics) do
			if not stealingAll then
				break
			end

			if info.Model and info.Model.Parent then
				local magicName = info.Name
				local originalModel = info.Model

				StatsLabel.Text = string.format(
					"Stealing %d / %d from %s: %s",
					index,
					total,
					selectedPlayer.Name,
					magicName
				)

				local stealSuccess, stealReason = stealOneMagic(originalModel)

				StatsLabel.Text = "Returning to collect zone after steal attempt: " .. magicName

				local returned, returnReason = returnToLocalPlayerBase()

				if returned then
					returnedToCollectZone += 1
				else
					warn("Failed to return to collect zone after steal:", magicName, returnReason)
				end

				task.wait(BETWEEN_ACTION_DELAY)

				if stealSuccess then
					stolen += 1

					local localStolenMagic = waitForLocalMagicByName(magicName, STEAL_CONFIRM_TIMEOUT)

					if isWhitelistedMagic(magicName) then
						skippedSelling += 1
						StatsLabel.Text = "Kept whitelisted magic: " .. magicName
					else
						StatsLabel.Text = "Selling stolen magic: " .. magicName

						local sellCandidate = localStolenMagic or findLocalMagicToSellByName(magicName, originalModel)

						if sellCandidate and sellCandidate.Parent then
							local sellSuccess, sellReason = sellOneMagic(sellCandidate)

							if sellSuccess then
								soldAfterSteal += 1
								StatsLabel.Text = "Sold stolen magic: " .. magicName
							else
								sellFailedAfterSteal += 1
								warn("Failed to sell stolen magic:", magicName, sellReason)
							end

							task.wait(BETWEEN_ACTION_DELAY)

							local returnedAgain, returnAgainReason = returnToLocalPlayerBase()

							if returnedAgain then
								returnedToCollectZone += 1
							else
								warn("Failed to return to collect zone after selling:", magicName, returnAgainReason)
							end
						else
							sellFailedAfterSteal += 1
							warn("Could not find stolen magic to sell:", magicName)
						end
					end
				else
					stealFailed += 1
					warn("Failed to steal:", magicName, stealReason)
				end

				task.wait(BETWEEN_MAGIC_DELAY)
			end
		end

		StatsLabel.Text = string.format(
			"Steal All done. Stolen: %d | Steal Failed: %d | Sold: %d | Sell Failed: %d | Kept: %d | Collect Returns: %d",
			stolen,
			stealFailed,
			soldAfterSteal,
			sellFailedAfterSteal,
			skippedSelling,
			returnedToCollectZone
		)
	end

	stealingAll = false
	updateStealAllButtonState()

	task.wait(BETWEEN_ACTION_DELAY)
	updateResults()
end

	return true, "protected"
end
