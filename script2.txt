local function n(value, fallback, minValue, maxValue)
	local number = tonumber(value)

	if number == nil then
		number = fallback
	end

	if minValue ~= nil then
		number = math.max(minValue, number)
	end

	if maxValue ~= nil then
		number = math.min(maxValue, number)
	end

	return number
end

local function clampVector(vec, maxMagnitude)
	if vec.Magnitude > maxMagnitude then
		return vec.Unit * maxMagnitude
	end

	return vec
end

local function isWhitelisted(plr)
	return plr and whitelist[plr.UserId] == true
end

local function isPlayerConsented(plr)
	if not REQUIRE_TARGET_CONSENT then
		return true
	end

	if not plr then
		return false
	end

	if plr == LocalPlayer then
		return true
	end

	if plr:GetAttribute(CONSENT_ATTRIBUTE) == true then
		return true
	end

	local character = plr.Character
	if character then
		if character:GetAttribute(CONSENT_ATTRIBUTE) == true then
			return true
		end

		if character:GetAttribute(TARGET_ATTRIBUTE) == true then
			return true
		end
	end

	return false
end

local function getPartsForPlayer(plr)
	if not plr then
		return nil, "No player"
	end

	if isWhitelisted(plr) then
		return nil, "Player is whitelisted"
	end

	if not isPlayerConsented(plr) then
		return nil, "Player has not opted in"
	end

	local character = plr.Character
	if not character then
		return nil, "No character"
	end

	local humanoid = character:FindFirstChild("Humanoid")
	local hrp = character:FindFirstChild("HumanoidRootPart")
	local stats = character:FindFirstChild("CharcterStats")

	if not humanoid then
		return nil, "Missing Humanoid"
	end

	if not hrp then
		return nil, "Missing HumanoidRootPart"
	end

	if not stats then
		return nil, "Missing CharcterStats"
	end

	return {
		Player = plr,
		Character = character,
		Humanoid = humanoid,
		HRP = hrp,
		Stats = stats,
	}, nil
end

local function fireServerFunction(functionName, args)
	if not MainEvent then
		return false, "MainEvent missing"
	end

	MainEvent:FireServer(REMOTE_CODE, functionName, args)
	return true
end

local function canOperate()
	if not allowedByStudio then
		return false, "Not running in Studio"
	end

	if not allowedByUserId then
		return false, "User not allowlisted"
	end

	if not mainOk or not MainEvent then
		return false, "Remotes not loaded"
	end

	return true
end

			return

			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end)

local function lightenColor(color, amount)
	return Color3.new(
		math.clamp(color.R + amount, 0, 1),
		math.clamp(color.G + amount, 0, 1),
		math.clamp(color.B + amount, 0, 1)
	)
end

local function setButtonActive(button, active, baseColor, activeColor)
	local highlightColor = activeColor or lightenColor(baseColor, 0.22)

	button.BackgroundColor3 = active and highlightColor or baseColor
	button.TextColor3 = Color3.fromRGB(255, 255, 255)

	local stroke = button:FindFirstChildOfClass("UIStroke")
	if stroke then
		stroke.Transparency = active and 0.15 or 1
		stroke.Color = active and lightenColor(highlightColor, 0.25) or baseColor
		stroke.Thickness = active and 2 or 1
	end
end

local function rebuildWhitelistPanel()
	for _, child in ipairs(whitelistList:GetChildren()) do
		if child:IsA("TextButton") then
			child:Destroy()
		end
	end

	local list = Players:GetPlayers()

	table.sort(list, function(a, b)
		return a.Name < b.Name
	end)

	for _, plr in ipairs(list) do
		local isWL = isWhitelisted(plr)

		local button = Instance.new("TextButton")
		button.Size = UDim2.new(1, -4, 0, 30)
		button.BackgroundColor3 = isWL and Color3.fromRGB(45, 120, 70) or Color3.fromRGB(120, 70, 45)
		button.Text = (isWL and "✓ " or "✕ ") .. plr.Name
		button.TextColor3 = Color3.fromRGB(255, 255, 255)
		button.TextSize = 12
		button.Font = Enum.Font.GothamBold
		button.AutoButtonColor = true
		button.Parent = whitelistList

		local corner = Instance.new("UICorner")
		corner.CornerRadius = UDim.new(0, 7)
		corner.Parent = button

		button.MouseButton1Click:Connect(function()
			if whitelist[plr.UserId] then
				whitelist[plr.UserId] = nil
				setStatus("Removed from whitelist: " .. plr.Name, Color3.fromRGB(120, 70, 45))
			else
				whitelist[plr.UserId] = true
				activeLoops[plr] = nil
				detectedTargets[plr] = nil
				setStatus("Added to whitelist: " .. plr.Name, Color3.fromRGB(45, 105, 65))
			end

			rebuildWhitelistPanel()
		end)
	end
end

local function getLoopInterval()
	local interval = n(intervalBox.Text, DEFAULTS.LoopInterval, CAPS.MinLoopInterval, CAPS.MaxLoopInterval)
	intervalBox.Text = tostring(interval)
	return interval
end

local function doDamage(parts)
	local amount = n(damageBox.Text, DEFAULTS.Damage, 0, CAPS.MaxDamage)
	damageBox.Text = tostring(amount)

	if amount <= 0 then
		return
	end

	fireServerFunction("Main_Damger_OC_R", {
		parts.Character,
		{
			D = amount,
			F = LocalPlayer.Character or parts.Character,
		},
	})
end

local function doALVC(parts)
	local vx = n(velXBox.Text, DEFAULTS.VelX, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vy = n(velYBox.Text, DEFAULTS.VelY, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vz = n(velZBox.Text, DEFAULTS.VelZ, -CAPS.MaxVelocity, CAPS.MaxVelocity)

	velXBox.Text = tostring(vx)
	velYBox.Text = tostring(vy)
	velZBox.Text = tostring(vz)

	local vec = clampVector(Vector3.new(vx, vy, vz), CAPS.MaxVelocity)

	fireServerFunction("ALVC", {
		parts.HRP,
		vec,
		{
			C = parts.Character,
		},
	})
end

local function startLoopForPlayer(plr)
	if activeLoops[plr] then
		return
	end

	if activeTest == "OFF" then
		return
	end

	if isWhitelisted(plr) then
		return
	end

	activeLoops[plr] = true

	local myNonce = loopNonce
	local myTest = activeTest

	task.spawn(function()
		while activeLoops[plr]
			and selectedZone
			and selectedZone.Parent
			and activeTest == myTest
			and myNonce == loopNonce
			and detectedTargets[plr]
			and not isWhitelisted(plr)
		do
			local parts = getPartsForPlayer(plr)

			if parts then
				if myTest == "Damage" then
					doDamage(parts)
				elseif myTest == "ALVC" then
					doALVC(parts)
				end
			end

			task.wait(getLoopInterval())
		end

		activeLoops[plr] = nil
	end)
end

local function stopLoopForPlayer(plr)
	if activeLoops[plr] then
		activeLoops[plr] = nil
	end
end

local function addResize(part)
	if activeHandles then
		activeHandles:Destroy()
	end

	local handles = Instance.new("Handles")
	handles.Adornee = part
	handles.Style = Enum.HandlesStyle.Resize
	handles.Color3 = Color3.fromRGB(255, 50, 50)
	handles.Parent = gui

	activeHandles = handles

	local startSize
	local startPosition
	local startFace

	handles.MouseButton1Down:Connect(function(face)
		startSize = part.Size
		startPosition = part.Position
		startFace = face
	end)

	handles.MouseDrag:Connect(function(face, distance)
		if face ~= startFace then
			return
		end

		local x = startSize.X
		local y = startSize.Y
		local z = startSize.Z

		local minSize = 2

		if face == Enum.NormalId.Right then
			x = math.max(minSize, startSize.X + distance)
			part.Position = startPosition + Vector3.new((x - startSize.X) / 2, 0, 0)
		elseif face == Enum.NormalId.Left then
			x = math.max(minSize, startSize.X - distance)
			part.Position = startPosition - Vector3.new((x - startSize.X) / 2, 0, 0)
		elseif face == Enum.NormalId.Top then
			y = math.max(minSize, startSize.Y + distance)
			part.Position = startPosition + Vector3.new(0, (y - startSize.Y) / 2, 0)
		elseif face == Enum.NormalId.Bottom then
			y = math.max(minSize, startSize.Y - distance)
			part.Position = startPosition - Vector3.new(0, (y - startSize.Y) / 2, 0)
		elseif face == Enum.NormalId.Front then
			z = math.max(minSize, startSize.Z + distance)
			part.Position = startPosition + Vector3.new(0, 0, (z - startSize.Z) / 2)
		elseif face == Enum.NormalId.Back then
			z = math.max(minSize, startSize.Z - distance)
			part.Position = startPosition - Vector3.new(0, 0, (z - startSize.Z) / 2)
		end

		part.Size = Vector3.new(x, y, z)
	end)
end

local function updatePlayerList()
	local allNames = {}
	local targetNames = {}
	local whitelistNames = {}

	for plr in pairs(detectedAll) do
		table.insert(allNames, plr.Name)
	end

	for plr in pairs(detectedTargets) do
		table.insert(targetNames, plr.Name)
	end

	for userId in pairs(whitelist) do
		local plr = Players:GetPlayerByUserId(userId)
		table.insert(whitelistNames, plr and plr.Name or tostring(userId))
	end

	table.sort(allNames)
	table.sort(targetNames)
	table.sort(whitelistNames)

	playerList.Text =
		"Inside: " .. (#allNames > 0 and table.concat(allNames, ", ") or "None") ..
		"\nTargets: " .. (#targetNames > 0 and table.concat(targetNames, ", ") or "None") ..
		"\nWhitelist: " .. (#whitelistNames > 0 and table.concat(whitelistNames, ", ") or "None")
end
local function updateZoneColor(zone, hasUnauthorized, hasTarget)
	if not zone or not zone.Parent then
		return
	end

	if hasUnauthorized then
		zone.Color = ZONE_ALERT_COLOR
	elseif hasTarget then
		zone.Color = ZONE_TARGET_COLOR
	else
		zone.Color = ZONE_IDLE_COLOR
	end
end
local function monitorZone(zone)
	task.spawn(function()
		while zone.Parent do
			local currentAll = {}
			local currentTargets = {}
            local hasUnauthorized = false

			local ok, parts = pcall(function()
				return workspace:GetPartsInPart(zone)
			end)

			if ok then
				for _, part in ipairs(parts) do
					local character = part:FindFirstAncestorOfClass("Model")

					if character and character:FindFirstChildOfClass("Humanoid") then
						local plr = Players:GetPlayerFromCharacter(character)

						if plr then
                            currentAll[plr] = true

                            if isPlayerConsented(plr) and not isWhitelisted(plr) then
                                currentTargets[plr] = true
                            elseif not isWhitelisted(plr) then
                                hasUnauthorized = true
                            end
                        end
					end
				end
			end

			for plr in pairs(detectedTargets) do
				if not currentTargets[plr] then
					stopLoopForPlayer(plr)
				end
			end

			detectedAll = currentAll
			detectedTargets = currentTargets

			for plr in pairs(detectedTargets) do
				startLoopForPlayer(plr)
			end
            local hasTarget = next(currentTargets) ~= nil
            updateZoneColor(zone, hasUnauthorized, hasTarget)

			local count = 0
			for _ in pairs(detectedTargets) do
				count += 1
			end

			if count > 0 then
				setStatus("Detected targets: " .. tostring(count), Color3.fromRGB(80, 45, 45))
			else
				if activeTest == "OFF" then
					setStatus("Ready. Zone active.", Color3.fromRGB(35, 35, 45))
				else
					setStatus("Waiting for non-whitelisted consented players.", Color3.fromRGB(45, 65, 100))
				end
			end

			updatePlayerList()

			task.wait(DEFAULTS.CheckInterval)
		end
	end)
end

local function createZone()
	if activeHandles then
		activeHandles:Destroy()
		activeHandles = nil
	end

	if selectedZone then
		selectedZone:Destroy()
		selectedZone = nil
	end

	loopNonce += 1
	table.clear(activeLoops)
	table.clear(detectedAll)
	table.clear(detectedTargets)

	local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	local root = character:WaitForChild("HumanoidRootPart")

	local zone = Instance.new("Part")
	zone.Name = "DebugZone"
	zone.Size = Vector3.new(20, 30, 75)
	zone.Position = root.Position
	zone.Anchored = true
	zone.CanCollide = false
	zone.CanTouch = true
	zone.CanQuery = true
	zone.Material = Enum.Material.Neon
	zone.Transparency = 0.9
	zone.Color = ZONE_IDLE_COLOR
	zone.Parent = workspace

	selectedZone = zone

	addResize(zone)
	monitorZone(zone)

	setStatus("Zone created.", Color3.fromRGB(45, 105, 65))
end

local function deleteZone()
	loopNonce += 1

	for plr in pairs(activeLoops) do
		stopLoopForPlayer(plr)
	end

	table.clear(activeLoops)
	table.clear(detectedAll)
	table.clear(detectedTargets)

	if activeHandles then
		activeHandles:Destroy()
		activeHandles = nil
	end

	if selectedZone then
		selectedZone:Destroy()
		selectedZone = nil
	end

	playerList.Text = "Inside: None\nTargets: None\nWhitelist: " .. LocalPlayer.Name

	setStatus("Zone deleted.", Color3.fromRGB(35, 35, 45))
end

spawnButton.MouseButton1Click:Connect(createZone)
deleteButton.MouseButton1Click:Connect(deleteZone)
