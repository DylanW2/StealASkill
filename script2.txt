local function n(value, fallback, minValue, maxValue)
	local number = tonumber(value)
	if number == nil then
		number = fallback
	end

	if minValue ~= nil then
		number = math.max(minValue, number)
	end

	if maxValue ~= nil then
		number = math.min(maxValue, number)
	end

	return number
end

local function clampVector(vec, maxMagnitude)
	if vec.Magnitude > maxMagnitude then
		return vec.Unit * maxMagnitude
	end

	return vec
end

local function isPlayerConsented(plr)
	if not REQUIRE_TARGET_CONSENT then
		return true
	end

	if not plr then
		return false
	end

	-- Let yourself test on yourself.
	if plr == LocalPlayer then
		return true
	end

	if plr:GetAttribute(CONSENT_ATTRIBUTE) == true then
		return true
	end

	local character = plr.Character
	if character then
		if character:GetAttribute(CONSENT_ATTRIBUTE) == true then
			return true
		end

		if character:GetAttribute(TARGET_ATTRIBUTE) == true then
			return true
		end
	end

	return false
end

local function getPartsForPlayer(plr)
	if not plr then
		return nil, "No player"
	end

	if not isPlayerConsented(plr) then
		return nil, "Player has not opted in"
	end

	local character = plr.Character
	if not character then
		return nil, "No character"
	end

	local humanoid = character:FindFirstChild("Humanoid")
	local hrp = character:FindFirstChild("HumanoidRootPart")
	local stats = character:FindFirstChild("CharcterStats")

	if not humanoid then
		return nil, "Missing Humanoid"
	end

	if not hrp then
		return nil, "Missing HumanoidRootPart"
	end

	if not stats then
		return nil, "Missing CharcterStats"
	end

	local stun = stats:FindFirstChild("Stun")
	local ragdoll = stats:FindFirstChild("IsRagdoll")
	local cantHit = stats:FindFirstChild("CantHit")

	if not stun then
		return nil, "Missing Stun"
	end

	if not ragdoll then
		return nil, "Missing IsRagdoll"
	end

	if not cantHit then
		return nil, "Missing CantHit"
	end

	return {
		Player = plr,
		Character = character,
		Humanoid = humanoid,
		HRP = hrp,
		Stats = stats,
		Stun = stun,
		Ragdoll = ragdoll,
		CantHit = cantHit,
	}, nil
end

local function fireServerFunction(functionName, args)
	if not MainEvent then
		return false, "MainEvent missing"
	end

	MainEvent:FireServer(REMOTE_CODE, functionName, args)
	return true
end

local function buildReport()
	local v, b, s, i = 0, 0, 0, 0

	for _, item in ipairs(resultData) do
		if item.State == "VULNERABLE" then
			v += 1
		elseif item.State == "BLOCKED" then
			b += 1
		elseif item.State == "SKIPPED" then
			s += 1
		elseif item.State == "INFO" then
			i += 1
		end
	end

	local lines = {}

	table.insert(lines, "Zone Security Probe Report")
	table.insert(lines, "========================================")
	table.insert(lines, "Runner: " .. LocalPlayer.Name .. " (" .. tostring(LocalPlayer.UserId) .. ")")
	table.insert(lines, "PlaceId: " .. tostring(game.PlaceId))
	table.insert(lines, "JobId: " .. tostring(game.JobId))
	table.insert(lines, "Studio: " .. tostring(RunService:IsStudio()))
	table.insert(lines, "Active Test: " .. tostring(activeTest))
	table.insert(lines, "")
	table.insert(lines, string.format("Summary: %d vulnerable, %d blocked, %d skipped, %d info.", v, b, s, i))
	table.insert(lines, "")

	for index, item in ipairs(resultData) do
		table.insert(lines, string.format("%02d. [%s] %s", index, item.State, item.Name))
		if item.Detail then
			table.insert(lines, "    " .. tostring(item.Detail))
		end
	end

	return table.concat(lines, "\n")
end

local function copyReport()
	local report = buildReport()

	if typeof(setclipboard) == "function" then
		local ok = pcall(function()
			setclipboard(report)
		end)

		if ok then
			setStatus("Report copied", Color3.fromRGB(45, 105, 65))
			return
		end
	end

	print(report)
	setStatus("Clipboard unavailable; printed report", Color3.fromRGB(70, 80, 110))
end

local function doDamage(parts)
	local amount = n(damageBox.Text, DEFAULTS.Damage, 0, CAPS.MaxDamage)
	damageBox.Text = tostring(amount)

	if amount <= 0 then
		return false, "damage is 0"
	end

	local before = parts.Humanoid.Health

	fireServerFunction("Main_Damger_OC_R", {
		parts.Character,
		{
			D = amount,
			F = LocalPlayer.Character or parts.Character,
		},
	})

	task.wait(0.08)

	local after = parts.Humanoid.Health

	if after < before - 0.01 then
		return true, string.format("Health %.2f -> %.2f", before, after)
	end

	return false, "health unchanged"
end

local function doStateCombo(parts)
	if parts.Ragdoll.Value == true then
		return false, "already ragdolled"
	end

	local stunValue = n(stunBox.Text, DEFAULTS.Stun, 0, CAPS.MaxStun)
	local cantHitValue = n(cantHitBox.Text, DEFAULTS.CantHit, 0, CAPS.MaxCantHit)
	local duration = n(ragTimeBox.Text, DEFAULTS.RagdollTime, 0, CAPS.MaxRagdollTime)
	local jumps = n(ragJumpBox.Text, DEFAULTS.RagdollJumps, 0, CAPS.MaxRagdollJumps)

	if parts.Stun.Value == stunValue then
		stunValue = stunValue == 1 and 2 or 1
	end

	if parts.CantHit.Value == cantHitValue then
		cantHitValue = math.clamp(cantHitValue + 1, 0, CAPS.MaxCantHit)
	end

	stunBox.Text = tostring(stunValue)
	cantHitBox.Text = tostring(cantHitValue)
	ragTimeBox.Text = tostring(duration)
	ragJumpBox.Text = tostring(jumps)

	fireServerFunction("ChangSTuNNNNNNN", {
		stunValue,
		parts.Character,
	})

	fireServerFunction("ChangCanTHiTTT", {
		cantHitValue,
		parts.Character,
	})

	fireServerFunction("Ragdoll_Character", {
		parts.Character,
		true,
		{
			T = duration,
			J = jumps,
		},
	})

	task.wait(0.12)

	local gotStun = parts.Stun.Value == stunValue
	local gotCantHit = parts.CantHit.Value == cantHitValue
	local gotRagdoll = parts.Ragdoll.Value == true

	local accepted = {}

	if gotStun then
		table.insert(accepted, "Stun")
	end

	if gotCantHit then
		table.insert(accepted, "CantHit")
	end

	if gotRagdoll then
		table.insert(accepted, "Ragdoll")
	end

	if #accepted > 0 then
		return true, "Accepted: " .. table.concat(accepted, ", ")
	end

	return false, "no state changed"
end

local function doALVC(parts)
	local vx = n(velXBox.Text, DEFAULTS.VelX, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vy = n(velYBox.Text, DEFAULTS.VelY, -CAPS.MaxVelocity, CAPS.MaxVelocity)
	local vz = n(velZBox.Text, DEFAULTS.VelZ, -CAPS.MaxVelocity, CAPS.MaxVelocity)

	velXBox.Text = tostring(vx)
	velYBox.Text = tostring(vy)
	velZBox.Text = tostring(vz)

	local vec = clampVector(Vector3.new(vx, vy, vz), CAPS.MaxVelocity)

	local before = parts.HRP.AssemblyLinearVelocity

	fireServerFunction("ALVC", {
		parts.HRP,
		vec,
		{
			C = parts.Character,
		},
	})

	task.wait(0.08)

	if (parts.HRP.AssemblyLinearVelocity - before).Magnitude > 4 then
		return true, "velocity changed"
	end

	return false, "velocity unchanged"
end

local function cleanupTarget(plr)
	local parts = getPartsForPlayer(plr)
	if not parts then
		return
	end

	fireServerFunction("ChangSTuNNNNNNN", {
		0,
		parts.Character,
	})

	fireServerFunction("ChangCanTHiTTT", {
		0,
		parts.Character,
	})

	fireServerFunction("Ragdoll_Character", {
		parts.Character,
		false,
	})
end

local function startLoopForPlayer(plr)
	if activeLoops[plr] then
		return
	end

	if activeTest == "OFF" then
		return
	end

	activeLoops[plr] = true

	local myNonce = loopNonce
	local myTest = activeTest
	local loggedVulnerable = false
	local loggedBlocked = false

	info("Loop Started", plr.Name .. " | " .. myTest)

	task.spawn(function()
		while activeLoops[plr]
			and selectedZone
			and selectedZone.Parent
			and activeTest == myTest
			and myNonce == loopNonce
			and detectedConsented[plr]
		do
			local parts, err = getPartsForPlayer(plr)
			if not parts then
				skipped("Loop " .. myTest, plr.Name .. ": " .. tostring(err))
				break
			end

			local ok = false
			local detail = nil

			if myTest == "Damage" then
				ok, detail = doDamage(parts)
			elseif myTest == "State x3" then
				ok, detail = doStateCombo(parts)
			elseif myTest == "ALVC" then
				ok, detail = doALVC(parts)
			else
				break
			end

			if ok and not loggedVulnerable then
				vuln("Loop " .. myTest, plr.Name .. ": " .. tostring(detail))
				loggedVulnerable = true
			elseif not ok and not loggedBlocked then
				blocked("Loop " .. myTest, plr.Name .. ": " .. tostring(detail))
				loggedBlocked = true
			end

			task.wait(getLoopInterval())
		end

		activeLoops[plr] = nil

		if myTest == "State x3" then
			cleanupTarget(plr)
		end

		info("Loop Stopped", plr.Name .. " | " .. myTest)
	end)
end

local function stopLoopForPlayer(plr)
	if activeLoops[plr] then
		activeLoops[plr] = nil
		cleanupTarget(plr)
	end
end

local function addResize(part)
	if activeHandles then
		activeHandles:Destroy()
	end

	local handles = Instance.new("Handles")
	handles.Adornee = part
	handles.Style = Enum.HandlesStyle.Resize
	handles.Color3 = Color3.fromRGB(255, 50, 50)
	handles.Parent = gui

	activeHandles = handles

	local startSize
	local startPosition
	local startFace

	handles.MouseButton1Down:Connect(function(face)
		startSize = part.Size
		startPosition = part.Position
		startFace = face
	end)

	handles.MouseDrag:Connect(function(face, distance)
		if face ~= startFace then
			return
		end

		local x = startSize.X
		local y = startSize.Y
		local z = startSize.Z

		local minSize = 2

		if face == Enum.NormalId.Right then
			x = math.max(minSize, startSize.X + distance)
			part.Position = startPosition + Vector3.new((x - startSize.X) / 2, 0, 0)
		elseif face == Enum.NormalId.Left then
			x = math.max(minSize, startSize.X - distance)
			part.Position = startPosition - Vector3.new((x - startSize.X) / 2, 0, 0)
		elseif face == Enum.NormalId.Top then
			y = math.max(minSize, startSize.Y + distance)
			part.Position = startPosition + Vector3.new(0, (y - startSize.Y) / 2, 0)
		elseif face == Enum.NormalId.Bottom then
			y = math.max(minSize, startSize.Y - distance)
			part.Position = startPosition - Vector3.new(0, (y - startSize.Y) / 2, 0)
		elseif face == Enum.NormalId.Front then
			z = math.max(minSize, startSize.Z + distance)
			part.Position = startPosition + Vector3.new(0, 0, (z - startSize.Z) / 2)
		elseif face == Enum.NormalId.Back then
			z = math.max(minSize, startSize.Z - distance)
			part.Position = startPosition - Vector3.new(0, 0, (z - startSize.Z) / 2)
		end

		part.Size = Vector3.new(x, y, z)
	end)
end

local function updatePlayerList()
	local allNames = {}
	local consentNames = {}

	for plr in pairs(detectedAll) do
		table.insert(allNames, plr.Name)
	end

	for plr in pairs(detectedConsented) do
		table.insert(consentNames, plr.Name)
	end

	table.sort(allNames)
	table.sort(consentNames)

	local allText = #allNames > 0 and table.concat(allNames, ", ") or "None"
	local consentText = #consentNames > 0 and table.concat(consentNames, ", ") or "None"

	playerList.Text =
		"Inside: " .. allText ..
		"\nConsented: " .. consentText ..
		"\nActive Test: " .. activeTest
end

local function monitorZone(zone)
	task.spawn(function()
		while zone.Parent do
			local currentAll = {}
			local currentConsented = {}

			local ok, parts = pcall(function()
				return workspace:GetPartsInPart(zone)
			end)

			if ok then
				for _, part in ipairs(parts) do
					local character = part:FindFirstAncestorOfClass("Model")

					if character and character:FindFirstChildOfClass("Humanoid") then
						local plr = Players:GetPlayerFromCharacter(character)

						if plr then
							currentAll[plr] = true

							if isPlayerConsented(plr) then
								currentConsented[plr] = true
							end
						end
					end
				end
			end

			for plr in pairs(detectedConsented) do
				if not currentConsented[plr] then
					stopLoopForPlayer(plr)
				end
			end

			detectedAll = currentAll
			detectedConsented = currentConsented

			for plr in pairs(detectedConsented) do
				startLoopForPlayer(plr)
			end

			local count = 0
			for _ in pairs(detectedConsented) do
				count += 1
			end

			if count > 0 then
				setStatus("Detected consented: " .. tostring(count), Color3.fromRGB(80, 45, 45))
			else
				if activeTest == "OFF" then
					setStatus("Ready. Zone active.", Color3.fromRGB(35, 35, 45))
				else
					setStatus("Waiting for consented players in zone.", Color3.fromRGB(45, 65, 100))
				end
			end

			updatePlayerList()

			task.wait(DEFAULTS.CheckInterval)
		end
	end)
end

local function createZone()
	if activeHandles then
		activeHandles:Destroy()
		activeHandles = nil
	end

	if selectedZone then
		selectedZone:Destroy()
		selectedZone = nil
	end

	loopNonce += 1
	table.clear(activeLoops)
	table.clear(detectedAll)
	table.clear(detectedConsented)

	local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	local root = character:WaitForChild("HumanoidRootPart")

	local zone = Instance.new("Part")
	zone.Name = "DebugZone"
	zone.Size = Vector3.new(20, 10, 20)
	zone.Position = root.Position
	zone.Anchored = true
	zone.CanCollide = false
	zone.CanTouch = true
	zone.CanQuery = true
	zone.Material = Enum.Material.Neon
	zone.Transparency = 0.3
	zone.Color = Color3.fromRGB(0, 200, 255)
	zone.Parent = workspace

	selectedZone = zone

	addResize(zone)
	monitorZone(zone)

	setStatus("Zone created.", Color3.fromRGB(45, 105, 65))
	info("Zone", "Created at " .. tostring(zone.Position))
end

local function deleteZone()
	loopNonce += 1

	for plr in pairs(activeLoops) do
		stopLoopForPlayer(plr)
	end

	table.clear(activeLoops)
	table.clear(detectedAll)
	table.clear(detectedConsented)

	if activeHandles then
		activeHandles:Destroy()
		activeHandles = nil
	end

	if selectedZone then
		selectedZone:Destroy()
		selectedZone = nil
	end

	playerList.Text = "Inside:\nNone"

	setStatus("Zone deleted.", Color3.fromRGB(35, 35, 45))
	info("Zone", "Deleted")
end
